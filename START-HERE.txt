<script type="module">
  import { supabase, requireAuth, wireSignOut, logActivity } from './js/supabaseClient.js';

  await requireAuth();
  wireSignOut();

  let profileId = null;
  let partnershipColumnAvailable = false;
  let currentSendingLimits = {};
  const DEFAULT_TARGET_MARKETS = ['UAE','Saudi Arabia','Qatar','Bahrain','Kuwait','Iran','India','Pakistan','International'];
  const DEFAULT_ROUTE_NOTES = 'Yachts Logistics provides shipping/logistics support from Shinas Port, Oman to Iran. Use this naturally for relevant Iranian, forwarding, trading and cargo prospects.';

  function linesToArray(text) {
    return text.split('\n').map(s => s.trim()).filter(Boolean);
  }

  function escapeHtml(value) {
    return String(value ?? '').replace(/[&<>"']/g, m => ({
      '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
    }[m]));
  }

  async function detectPartnershipColumn() {
    const result = await supabase.from('company_profile')
      .select('operational_partnership_model')
      .limit(1);

    partnershipColumnAvailable = !result.error;
    const field = document.getElementById('operational_partnership_model');
    const note = document.getElementById('partnership-field-note');

    if (!partnershipColumnAvailable) {
      field.disabled = true;
      note.innerHTML = 'Run the included <code>sql/add-operational-partnership-model.sql</code> once in Supabase to enable this field. The rest of Settings continues to work normally.';
    } else {
      field.disabled = false;
      note.textContent = 'This is kept separate from Services so AI workers do not confuse a partnership structure with a service you sell.';
    }
  }

  async function loadProfile() {
    await detectPartnershipColumn();

    const { data, error } = await supabase.from('company_profile').select('*').limit(1);
    if (error) {
      document.getElementById('settings-error').textContent = error.message;
      return;
    }
    if (!data || data.length === 0) return;

    const p = data[0];
    profileId = p.id;
    document.getElementById('services').value = (p.services || []).join('\n');
    document.getElementById('ports_covered').value = (p.ports_covered || []).join('\n');
    document.getElementById('approved_claims').value = (p.approved_claims || []).join('\n');
    document.getElementById('forbidden_claims').value = (p.forbidden_claims || []).join('\n');
    document.getElementById('signature').value = p.signature || '';
    currentSendingLimits = (p.sending_limits && typeof p.sending_limits === 'object') ? p.sending_limits : {};
    document.getElementById('daily_max').value = currentSendingLimits.daily_max ?? 20;
    document.getElementById('ai_openai_enabled').checked = currentSendingLimits.ai_openai_enabled !== false;
    document.getElementById('openai_monthly_budget_usd').value = Math.min(4.5, Number(currentSendingLimits.openai_monthly_budget_usd ?? 4.5));
    document.getElementById('gemini_daily_request_limit').value = Math.min(24, Math.max(1, parseInt(currentSendingLimits.gemini_daily_request_limit ?? 20, 10)));
    document.getElementById('target_markets').value = Array.isArray(currentSendingLimits.target_markets)
      ? currentSendingLimits.target_markets.join('\n')
      : DEFAULT_TARGET_MARKETS.join('\n');
    document.getElementById('outreach_route_notes').value = currentSendingLimits.outreach_route_notes || DEFAULT_ROUTE_NOTES;
    updateAiModeLine();

    if (partnershipColumnAvailable) {
      document.getElementById('operational_partnership_model').value = p.operational_partnership_model || '';
    }
  }

  function updateAiModeLine() {
    const on = document.getElementById('ai_openai_enabled').checked;
    const line = document.getElementById('ai-mode-line');
    line.textContent = on ? 'OpenAI Luna preferred · Gemini automatic fallback' : 'OpenAI paused · Gemini active';
    line.className = 'ai-mode-line ' + (on ? 'state-ok' : 'state-warn');
  }

  document.getElementById('ai_openai_enabled').addEventListener('change', updateAiModeLine);

  document.getElementById('settings-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    const errorBox = document.getElementById('settings-error');
    const btn = document.getElementById('settings-save');
    errorBox.textContent = '';
    btn.disabled = true;

    const payload = {
      services: linesToArray(document.getElementById('services').value),
      ports_covered: linesToArray(document.getElementById('ports_covered').value),
      approved_claims: linesToArray(document.getElementById('approved_claims').value),
      forbidden_claims: linesToArray(document.getElementById('forbidden_claims').value),
      signature: document.getElementById('signature').value.trim(),
      sending_limits: {
        ...currentSendingLimits,
        daily_max: parseInt(document.getElementById('daily_max').value || '20', 10),
        ai_openai_enabled: document.getElementById('ai_openai_enabled').checked,
        openai_monthly_budget_usd: Math.min(4.5, Math.max(0.1, parseFloat(document.getElementById('openai_monthly_budget_usd').value || '4.5'))),
        gemini_daily_request_limit: Math.min(24, Math.max(1, parseInt(document.getElementById('gemini_daily_request_limit').value || '20', 10))),
        target_markets: linesToArray(document.getElementById('target_markets').value),
        outreach_route_notes: document.getElementById('outreach_route_notes').value.trim()
      }
    };

    if (partnershipColumnAvailable) {
      payload.operational_partnership_model =
        document.getElementById('operational_partnership_model').value.trim();
    }

    let error;
    if (profileId) {
      ({ error } = await supabase.from('company_profile').update(payload).eq('id', profileId));
    } else {
      const { data, error: insertError } = await supabase.from('company_profile').insert(payload).select().single();
      error = insertError;
      if (data) profileId = data.id;
    }

    btn.disabled = false;
    if (error) {
      errorBox.textContent = error.message;
      return;
    }

    currentSendingLimits = payload.sending_limits;
    await logActivity('Manual Entry', 'Updated company settings', 'company_profile', profileId, {
      operational_partnership_model_saved: partnershipColumnAvailable,
      ai_openai_enabled: payload.sending_limits.ai_openai_enabled,
      openai_monthly_budget_usd: payload.sending_limits.openai_monthly_budget_usd,
      gemini_daily_request_limit: payload.sending_limits.gemini_daily_request_limit
    });
    btn.textContent = 'Saved ✓';
    setTimeout(() => { btn.textContent = 'Save settings'; }, 1800);
  });

  await loadProfile();
</script>
